import numpy as np

def find_matches(selfie, faces, threshold=0.5):
    """Return indexes of faces that look like the selfie."""
    selfie = selfie / np.linalg.norm(selfie)
    faces = faces / np.linalg.norm(faces, axis=1, keepdims=True)
    scores = faces @ selfie
    return [i for i, score in enumerate(scores)
            if score >= threshold]
